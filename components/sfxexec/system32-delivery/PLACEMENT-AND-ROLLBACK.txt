BEGIN PLACE-BODY
$ErrorActionPreference = 'Stop'
function Read-SfxDigest($stream) {
    $stream.Position=0; $sha=[Security.Cryptography.SHA256]::Create()
    try { return [BitConverter]::ToString($sha.ComputeHash($stream)).Replace('-','') }
    finally { $sha.Dispose(); $stream.Position=0 }
}
foreach($p in @($source,$target,$receipt)) {
    if(-not [IO.Path]::IsPathRooted($p) -or $p.StartsWith('\\') -or $p -match "[`r`n]"){throw 'Absolute local paths required'}
    $walk=[IO.Path]::GetFullPath($p)
    while($walk){if([IO.File]::Exists($walk) -or [IO.Directory]::Exists($walk)){if(([IO.File]::GetAttributes($walk) -band [IO.FileAttributes]::ReparsePoint) -ne 0){throw 'Redirected path refused'}};$walk=[IO.Path]::GetDirectoryName($walk)}
}
if([IO.Path]::GetFileName($target) -cne 'SFXEXEC.DLL' -or $source -ieq $target -or $receipt -ieq $target -or $receipt -ieq $source){throw 'Invalid one-file paths'}
$inputFile=[IO.File]::Open($source,[IO.FileMode]::Open,[IO.FileAccess]::Read,[IO.FileShare]::Read)
$targetFile=$null; $record=$null
try {
    if((Read-SfxDigest $inputFile) -cne $expected){throw 'Source hash differs from approved candidate'}
    $existed=[IO.File]::Exists($target)
    if($existed){$targetFile=[IO.File]::Open($target,[IO.FileMode]::Open,[IO.FileAccess]::Read,[IO.FileShare]::Read)
        if((Read-SfxDigest $targetFile) -cne $expected){throw 'CONFLICT: existing SFXEXEC.DLL differs; preserved. Ask for reviewed update plan'}}
    $record=[IO.File]::Open($receipt,[IO.FileMode]::CreateNew,[IO.FileAccess]::ReadWrite,[IO.FileShare]::Read)
    $pending=[Text.Encoding]::UTF8.GetBytes("SFX-System32-1`n$target`n$expected`npending`nfalse`n")
    $record.Write($pending,0,$pending.Length);$record.Flush()
    if(-not $existed){

        $buffer=New-Object byte[] 16384

        $targetFile.Flush()

} finally {if($targetFile){$targetFile.Dispose()};if($record){$record.Dispose()};$inputFile.Dispose()}
